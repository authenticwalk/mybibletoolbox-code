# Directory Mapping
 - this folder is $ROOT/bible-study-tools/tbta/features
 - the data director is $DATA_DIR which defaults to $ROOT/.data
 - if your CWD is /workspace and file is /User/.../bible-study-tools replace that prefix with /workspace as you are in docker
 - $TBTA-DIR is $ROOT/bible-study-tools/tbta
 - $FEATURES-DIR is $ROOT/bible-study-tools/tbta/features
 - $CURRENT-FEATURE-DIR is $ROOT/bible-study-tools/tbta/features/${feature}
